import 'package:flutter/material.dart';

class CaixaTexto extends StatelessWidget {
  // construtor
  const CaixaTexto({
    super.key,
  });

  @override
  Widget build(BuildContext context) {
    return Container(
      decoration: const BoxDecoration(
        gradient: LinearGradient(
          begin: Alignment.topCenter,
          colors: [
            Colors.red,
            Colors.lightBlue,
          ],
          end: Alignment.bottomCenter,
        ),
      ),
      child: Center(
        child: TextoComEstilo(
          'Hello Barretos!',
          tamanho: 36,
        ),
      ),
    );
  }
}

class TextoComEstilo extends StatelessWidget {
  TextoComEstilo(
    this.mostrar, {
    required this.tamanho,
    super.key,
  });

  TextoComEstilo.ifsp({super.key})
      : mostrar = "Hello, IFSP!",
        tamanho = 42;

  String mostrar;
  double tamanho;

  @override
  Widget build(BuildContext context) {
    return Text(
      mostrar,
      style: TextStyle(
        fontSize: tamanho,
        color: Colors.black,
      ),
    );
  }
}
